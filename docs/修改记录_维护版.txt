七、v1.137怪物信息控制器源码（GBK编码后放入Sys.lua；v1.138沿用）
-- Monster tooltip coordinator.  Existing monster callbacks still format the
-- name, level, elements and icons; this only selects/refreshes the view.
local marker = '__gm_monsterinfo_v137'
local statekey = '__gm_monsterinfo_state_v137'
local errkey = '__gm_monsterinfo_error_v137'
local ownedkey = '__gm_monsterinfo_owned_v137'
local category = '__gm_monsterinfo_v137'
local noop = function(_) end
local hide = function(a) if type(a) == 'table' then a['消失'] = true end end

local function report(C, why)
  local message = string.sub(tostring(why), 1, 2048)
  if C[errkey] == message then return end
  C[errkey] = message
  local lib = _G.io
  if type(lib) ~= 'table' or type(lib.open) ~= 'function' then return end
  local ok, file = pcall(lib.open, 'Log/guimeng_monsterinfo_error.txt', 'a')
  if not ok or not file then return end
  pcall(function() file:write('v1.137: ', message, '\n'); file:close() end)
end

local function tick(C)
  if type(C) ~= 'table' then return end
  local setting = C['script_setting']
  local U = type(setting) == 'table' and setting['Ux雌'] or nil
  local E = _G['Ae冻集']
  local S = _G['Sys']
  local objects, bubbles = C['屏幕事件组'], C['屏幕气泡组']
  if type(U) ~= 'table' or type(E) ~= 'table' or type(S) ~= 'table'
      or type(S.run) ~= 'function' or type(objects) ~= 'table'
      or type(bubbles) ~= 'table' then return end

  local state = C[statekey]
  if type(state) ~= 'table' then state = {}; C[statekey] = state end

  local function valid(o)
    if type(o) ~= 'table' or o.id == nil or o['已经消失'] or o['不可选中']
        or o['是否有效'] == false or o['名称显示方式'] == 0 then return false end
    local scripts, a = o['回调脚本'], o['属性']
    if type(scripts) ~= 'table' or type(scripts[marker]) ~= 'function'
        or type(scripts['动作监视']) ~= 'function' or type(a) ~= 'table' then return false end
    local hp = tonumber(a.hp)
    if not hp or hp ~= hp or hp <= 0 then return false end
    for i=1,#objects do if objects[i] == o then return true end end
    return false
  end

  local function getbubble(list)
    for i=#list,1,-1 do
      local b = list[i]
      local o = type(b) == 'table' and b['对象'] or nil
      local chat = type(o) == 'table' and o['聊天内容组'] or nil
      local a = type(chat) == 'table' and chat['属性回调'] or nil
      local text = type(a) == 'table' and a['内容'] or nil
      if type(text) == 'string' and string.find(text,'#t(4,4,4,4)',1,true)
          and string.find(text,'攻击元素',1,true) and string.find(text,'防御元素',1,true) then
        return b, o, a
      end
    end
  end

  local function destroy(b)
    local object = type(b)=='table' and b['对象'] or nil
    local f = type(object)=='table' and object['销毁'] or nil
    if type(f)=='function' then pcall(f, object) end
  end
  local function remove_old(keep)
    for i=#bubbles,1,-1 do
      local b=bubbles[i]
      if b~=keep and type(b)=='table' and b[ownedkey] then
        destroy(b); table.remove(bubbles,i)
      end
    end
  end

  local hover = E['鼠标焦点事件对象']
  if not valid(hover) then hover = nil end
  local control = C['控制对象']
  local selected
  if type(control) == 'table' then
    local physical, ranged = control['攻击目标'], control['远程目标']
    if valid(physical) then selected = physical
    elseif valid(ranged) then selected = ranged end
  end
  local desired = hover or selected
  local hero = C.Hero
  local ha = type(hero) == 'table' and hero['属性'] or nil
  local hhp = type(ha) == 'table' and tonumber(ha.hp) or nil
  if hhp and hhp <= 0 then desired = nil; hover = nil end

  local b, rich, a = getbubble(bubbles)
  if b ~= state.bubble then
    state.bubble = b
    state.owner = (b and hover and U['怪物描述'] == hover.id) and hover or nil
  end

  if not desired then
    if a then a['消失'] = true; rich.fun = hide end
    U['怪物描述'] = nil
    state.bubble, state.owner = nil, nil
    return
  end

  if b and state.owner == desired and rich['可视'] ~= false
      and rich['动感进度'] ~= 2 then
    -- Reuse the current view and avoid creating sprites every frame.
  else
    -- Invoke only a positively marked native monster tooltip callback.
    -- Both shared fields are restored even if the callback fails.
    local temporary = {}
    local oldid, olddesc = U['鼠标事件ID'], U['怪物描述']
    U['鼠标事件ID'], U['怪物描述'] = desired.id, nil
    local scripts = desired['回调脚本']
    -- Native category 0 creation normally deletes other category 0 bubbles.
    -- Create in a private list, then merge only our bubble into the real list.
    C['屏幕气泡组'] = temporary
    local ok, why = pcall(S.run, 'fun', scripts['动作监视'], {0,0})
    C['屏幕气泡组'] = bubbles
    U['鼠标事件ID'], U['怪物描述'] = oldid, olddesc
    local made, made_rich, made_a = getbubble(temporary)
    if not ok or not made then
      for i=1,#temporary do destroy(temporary[i]) end
      if a then a['消失'] = true; rich.fun = hide end
      U['怪物描述'] = nil
      state.bubble, state.owner = nil, nil
      if not ok then report(C, why) end
      return
    end
    for i=1,#temporary do if temporary[i]~=made then destroy(temporary[i]) end end
    b, rich, a = made, made_rich, made_a
    b['分类'], b[ownedkey] = category, true
    table.insert(bubbles,b)
    state.bubble, state.owner = b, desired
  end

  -- Original callbacks dismiss the bubble solely on mouse movement. The
  -- coordinator now owns that lifetime and retains the selected target.
  a['消失'] = false
  rich.fun = noop
  b['分类'], b[ownedkey] = category, true
  remove_old(b)
  U['怪物描述'] = hover and hover.id or nil

  -- Read the same live native HP fields used by the original blood bar. Only
  -- replace the numeric span; preserve all name/element/icon formatting.
  local attr = desired['属性']
  local hp, maxhp = tonumber(attr.hp), tonumber(attr.maxhp)
  local text = a['内容']
  if hp and maxhp and hp == hp and maxhp == maxhp and type(text) == 'string' then
    local fresh = string.gsub(text, '(%)) [^ ]+( 攻击元素:#W)',
      function(left,right) return left..' '..tostring(hp)..'/'..tostring(maxhp)..right end, 1)
    if fresh ~= text then a['内容'] = fresh end
  end
end

return function(C)
  if type(pcall) ~= 'function' then return end
  local ok, why = pcall(tick, C)
  if not ok and type(C) == 'table' then pcall(report, C, why) end
end
